  <script>

    (function() {

      let timerId;

      const QUESTIONS = [

        { text: "Quel est le nom de l’unité utilisée pour exprimer l’intensité d’une force ?", options: [

            { id: 'a', text: 'Le kilogramme' },

            { id: 'b', text: 'Le mètre' },

            { id: 'c', text: 'Le newton', correct: true },

            { id: 'd', text: 'Le volt' }

          ], explanation: 'L’unité SI de la force est le newton (N).' },

        { text: "Quand on parle de force, de quoi s'agit-il ?", options: [

            { id: 'a', text: 'Une action mécanique qui modifie un mouvement ou déforme un objet', correct: true },

            { id: 'b', text: 'La vitesse d’un objet en mouvement' },

            { id: 'c', text: 'Le poids d’unobjet' },

            { id: 'd', text: 'L’énergie contenue dans un objet' }

          ], explanation: 'Une force modifie le mouvement ou la forme d’un objet.' },

        { text: "Quel instrument mesure une force ?", options: [

            { id: 'a', text: 'Dynamomètre', correct: true },

            { id: 'b', text: 'Balance' },

            { id: 'c', text: 'Thermomètre' },

            { id: 'd', text: 'Baromètre' }

          ], explanation: 'Le dynamomètre mesure directement la force appliquée.' },

        { text: "Quelle est la formule mathématique de la force résultante sur un objet ?", options: [

            { id: 'a', text: 'F = m * a', correct: true },

            { id: 'b', text: 'F = m / a' },

            { id: 'c', text: 'F = a / m' },

            { id: 'd', text: 'F = m + a' }

          ], explanation: 'F = m × a (2ᵉ loi de Newton).' },

        { text: "Quelle unité correspond à 1 N dans le système SI ?", options: [

            { id: 'a', text: '1 kg·m/s²', correct: true },

            { id: 'b', text: '1 kg/m²' },

            { id: 'c', text: '1 m/s' },

            { id: 'd', text: '1 J' }

          ], explanation: '1 N = 1 kg·m/s².' }

      ];

      let userName = '',

          questions = [],

          current = 0,

          totalScore = 0,

          totalTime = 0,

          answersDetail = [];

      const attempts = JSON.parse(localStorage.getItem('quizAttempts') || '[]');

      const app = document.getElementById('app');

      // Couleurs pour Chart.js

      const fgColor = getComputedStyle(document.documentElement).getPropertyValue('--fg').trim();

      const primaryColor = getComputedStyle(document.documentElement).getPropertyValue('--primary').trim();

 

      const correctMsgs = ['Bien joué !', 'Bravo !', 'Chapeau !'];

      const wrongMsgs = ['Oups...', 'Dommage !', 'Pas tout à fait...'];

      const randMsg = arr => arr[Math.floor(Math.random() * arr.length)];

 

      function shuffle(arr) {

        const a = [...arr];

        for (let i = a.length - 1; i > 0; i--) {

          const j = Math.floor(Math.random() * (i + 1));

          [a[i], a[j]] = [a[j], a[i]];

        }

        return a;

      }

      function saveAttempt() {

        attempts.push({ num: attempts.length + 1, score: totalScore });

        localStorage.setItem('quizAttempts', JSON.stringify(attempts));

      }

 

      function renderStart() {

        app.innerHTML = `

          <h1>Quiz sur les forces</h1>

          <input id="name-in" placeholder="Ton prénom" value="${userName}" />

          <button class="primary" id="start-btn">Démarrer</button>

          ${attempts.length ? `<p>Joué ${attempts.length} fois</p>` : ''}

          <small>Créé par eSCIENCES avec beaucoup de passion et de café</small>

        `;

        document.getElementById('start-btn').onclick = () => {

          const v = document.getElementById('name-in').value.trim();

          if (!v) return;

          userName = v;

          startCountdown();

        };

      }

 

      function startCountdown() {

        questions = shuffle(QUESTIONS.map(q => ({ ...q, options: shuffle(q.options) })));

        current = 0;

        totalScore = 0;

        totalTime = 0;

        answersDetail = [];

        showCountdown(3);

      }

 

      function showCountdown(n) {

        if (n > 0) {

          app.innerHTML = `<div class="countdown" data-value="${n}">${n}</div>`;

          setTimeout(() => showCountdown(n - 1), 1000);

        } else {

          app.innerHTML = `<div class="countdown">Go !</div>`;

          setTimeout(renderQuiz, 1000);

        }

      }

 

      function renderQuiz() {

        const DURATION = 30;

        const startTime = performance.now();

        cancelAnimationFrame(timerId);

        app.innerHTML = `

          <div class="progress-bar"><div class="progress" id="bar"></div></div>

          <h2>${questions[current].text}</h2>

          <div id="opts">

            ${questions[current].options.map(o => `<div class="question" data-id="${o.id}">${o.text}</div>`).join('')}

          </div>

          <div id="feedback"></div>

        `;

        document.querySelectorAll('.question').forEach(el => el.onclick = () => handleAnswer(el.dataset.id, startTime));

        (function tick() {

          const elapsed = (performance.now() - startTime) / 1000;

          const rem = Math.max(0, DURATION - elapsed);

          document.getElementById('bar').style.width = (rem / DURATION * 100) + '%';

          if (rem > 0) timerId = requestAnimationFrame(tick);

        })();

      }

 

      function handleAnswer(id, startTime) {

        cancelAnimationFrame(timerId);

        const DURATION = 30;

        const elapsed = (performance.now() - startTime) / 1000;

        totalTime += Math.min(elapsed, DURATION);

        const timeLeft = Math.max(0, DURATION - elapsed);

        const q = questions[current];

        const correctOpt = q.options.find(o => o.correct);

        const earned = id === correctOpt.id ? Math.floor(timeLeft / DURATION * 1000) : 0;

        totalScore += earned;

        answersDetail.push({ question: q.text, chosen: q.options.find(o => o.id === id)?.text || 'Aucune', correct: correctOpt.text, earned, explanation: q.explanation });

        document.querySelectorAll('.question').forEach(el => {

          if (el.dataset.id === correctOpt.id) el.classList.add('correct');

          if (el.dataset.id === id && id !== correctOpt.id) el.classList.add('wrong');

        });

        const ok = id === correctOpt.id;

        const msg = ok ? randMsg(correctMsgs) : randMsg(wrongMsgs);

        document.getElementById('feedback').innerHTML = `<p>${ok ? '✅' : '❌'} ${msg} +${earned} pts</p><button class="primary" id="next-btn">Suivant →</button>`;

        document.getElementById('next-btn').onclick = () => {

          if (current + 1 < questions.length) {

            current++;

            renderQuiz();

          } else {

            saveAttempt();

            renderSummary();

          }

        };

      }

 

      function renderSummary() {

        const correctCount = answersDetail.filter(a => a.earned > 0).length;

        const maxScore = QUESTIONS.length * 1000;

        const pct = Math.round(totalScore / maxScore * 100);

        app.innerHTML = `

          <h2>Résumé du Quiz</h2>

          <div class="summary-cards">

            <div class="card"><h3>${correctCount}</h3><p>Bonnes réponses</p></div>

            <div class="card"><h3>${pct}%</h3><p>Score final</p></div>

            <div class="card"><h3>${Math.round(totalTime)}</h3><p>Temps total</p></div>

          </div>

          <button class="primary" id="retry-btn">Rejouer</button>

          <button class="secondary" id="video-btn">Regarder la vidéo 🎥</button>

          <h3>Le récapitulatif de tes scores</h3>

          <canvas id="chart" style="margin-top:1rem;"></canvas>

          <div class="recap">

            <h3>Détails des réponses</h3>

            <ul>

              ${answersDetail.map((a, i) => `<li class="${a.earned > 0 ? 'correct' : 'wrong'}"><strong>${i+1}. ${a.question}</strong><p>Ta réponse : ${a.chosen}</p>${a.earned === 0 ? `<p>Bonne réponse : ${a.correct}</p>` : ''}<p>Points : ${a.earned}</p><p><em>Explication : ${a.explanation}</em></p></li>`).join('')}

            </ul>

          </div>

        `;

        const ctx = document.getElementById('chart').getContext('2d');

        new Chart(ctx, {

          type: 'line',

          data: {

            labels: attempts.map(a => a.num),

            datasets: [{ data: attempts.map(a => a.score), borderColor: primaryColor, fill: false, tension: 0.4, pointBackgroundColor: fgColor }]

          },

          options: {

            scales: {

              x: { grid: { display: true, color: 'rgba(0,0,0,0.2)' }, title: { display: true, text: 'Parties jouées' } },

              y: { beginAtZero: true, max: maxScore, grid: { display: true, color: 'rgba(0,0,0,0.2)' } }

            },

            plugins: { legend: { display: false } }

          }

        });

        document.getElementById('retry-btn').onclick = startCountdown;

        document.getElementById('video-btn').onclick = () => window.open('https://youtu.be/dQw4w9WgXcQ', '_blank');

      }

 

      renderStart();

    })();

  </script>
